```sql
SELECT 
    p.nome_pessoa AS diretor,
    COUNT(DISTINCT bp.sk_movie_id) AS filmes,

JOIN bridge_movie_person bp ON bp.sk_person_id = p.sk_person_id
```

SQL gerado:
```sql
SELECT a.nome_pessoa AS ator, d.nome_pessoa AS diretor, COUNT(DISTINCT bp1.sk_movie_id) AS filmes
FROM bridge_movie_person bp1
JOIN dim_people a ON a.sk_person_id = bp1.sk_person_id AND a.tipo_pessoa = 'Ator'
JOIN bridge_movie_person bp2 ON bp2.sk_movie_id = bp1.sk_movie_id
JOIN dim_people d ON d.sk_person_id = bp2.sk_person_id AND d.tipo_pessoa = 'Diretor'

GROUP BY a.sk_person_id, d.sk_person_id
ORDER BY filmes DESC
LIMIT 10
```
